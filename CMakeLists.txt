cmake_minimum_required(VERSION 3.24)
project(QuickStation VERSION 1.6.14 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "QuickStation requires 64-bit Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /utf-8 /EHsc /guard:cf)
add_compile_options($<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/DNDEBUG> $<$<CONFIG:Release>:/Gy> $<$<CONFIG:Release>:/Gw>)
add_link_options(/DYNAMICBASE /NXCOMPAT /guard:cf)
add_link_options($<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF> $<$<CONFIG:Release>:/INCREMENTAL:NO>)
add_library(radio_core src/core.cpp src/storage.cpp src/directory.cpp)
target_include_directories(radio_core PUBLIC src third_party)
target_link_libraries(radio_core PUBLIC shell32 ole32)
add_library(radio_audio src/audio.cpp)
target_include_directories(radio_audio PUBLIC third_party/bass/c)
target_link_libraries(radio_audio PUBLIC radio_core)
add_executable(QuickStationAudio WIN32 src/audio_worker.cpp)
target_include_directories(QuickStationAudio PRIVATE third_party/bass/c)
target_link_libraries(QuickStationAudio PRIVATE radio_core "${CMAKE_SOURCE_DIR}/third_party/bass/c/x64/bass.lib")
set(QUICKSTATION_UI_SOURCES
  src/app.cpp src/ui_metrics.cpp src/button_icons.cpp src/window_state.cpp
  src/browse_dialog.cpp src/logos.cpp src/network.cpp assets/radio.rc)
add_executable(QuickStation WIN32 src/main.cpp ${QUICKSTATION_UI_SOURCES})
target_link_libraries(QuickStation PRIVATE radio_core radio_audio winhttp comctl32 comdlg32 windowscodecs uxtheme)
target_link_options(QuickStation PRIVATE /MANIFEST:NO)
add_dependencies(QuickStation QuickStationAudio)
foreach(target QuickStation QuickStationAudio)
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/bass/x64/bass.dll" "$<TARGET_FILE_DIR:${target}>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/basshls/x64/basshls.dll" "$<TARGET_FILE_DIR:${target}>")
endforeach()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE radio_core)
add_test(NAME core COMMAND core_tests)
add_executable(audio_tests tests/audio_tests.cpp)
target_link_libraries(audio_tests PRIVATE radio_audio ws2_32)
add_test(NAME audio COMMAND audio_tests)
set_tests_properties(audio PROPERTIES TIMEOUT 150)
add_dependencies(audio_tests QuickStationAudio)
add_executable(logo_tests tests/logo_tests.cpp src/logos.cpp)
target_link_libraries(logo_tests PRIVATE radio_core windowscodecs comctl32)
add_test(NAME logos COMMAND logo_tests)
add_executable(native_ui_tests tests/native_ui_tests.cpp ${QUICKSTATION_UI_SOURCES})
target_compile_definitions(native_ui_tests PRIVATE QUICKSTATION_UI_TESTS)
target_link_libraries(native_ui_tests PRIVATE radio_core radio_audio winhttp comctl32 comdlg32 windowscodecs uxtheme)
target_link_options(native_ui_tests PRIVATE /MANIFEST:NO)
add_test(NAME native_ui COMMAND native_ui_tests)
set_tests_properties(native_ui PROPERTIES TIMEOUT 150)
add_executable(directory_tests tests/directory_tests.cpp src/network.cpp)
target_link_libraries(directory_tests PRIVATE radio_core winhttp ws2_32 windowscodecs)
add_test(NAME directory COMMAND directory_tests)
set_tests_properties(directory PROPERTIES TIMEOUT 45)
